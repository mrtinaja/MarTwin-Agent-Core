    CABECERA: `
      SELECT
        OP.CAOS_CD_USUARIO AS "USUARIO|string",
        U.CAUS_CAGR_CD_COMPANIA AS "CD_COMPANIA|string",
        U.CAUS_CAGR_CD_UNIDAD_ADM AS "CD_UNIDAD_ADM|string",
        G.CAGR_NM_UNIDAD_ADM AS "NM_UNIDAD_ADM|string",
        TO_CHAR(TRUNC(LEAST(OP.CAOS_FE_A, OP.CAOS_FE_B)),'DD-MM-YYYY') AS "FECHA_DESDE|string",
        TO_CHAR(TRUNC(GREATEST(OP.CAOS_FE_A, OP.CAOS_FE_B)),'DD-MM-YYYY') AS "FECHA_HASTA|string",
        TO_CHAR(SYSDATE,'DD-MM-YYYY') AS "FECHA_ACTUAL|string",
        TO_CHAR(SYSDATE,'HH24:MI:SS') AS "HORA_ACTUAL|string"
      FROM CART_ORDENES_PROCESOS OP
      JOIN CART_USUARIOS U
        ON U.CAUS_CD_USUARIO = OP.CAOS_CD_USUARIO
      JOIN CART_GERENCIAS G
        ON G.CAGR_CD_COMPANIA   = U.CAUS_CAGR_CD_COMPANIA
       AND G.CAGR_CD_UNIDAD_ADM = U.CAUS_CAGR_CD_UNIDAD_ADM
      WHERE OP.CAOS_NU_SECUENCIA = :NU_SECUENCIA
    `,

    DATOS_COMPANIA: `
      SELECT
        C.CJCM_NM_COMPANIA AS "NM_COMPANIA|string",
        'CUIT:' ||
          SUBSTR(LPAD(TRIM(C.CJCM_NU_CUIT),11,'0'),1,2) || '-' ||
          SUBSTR(LPAD(TRIM(C.CJCM_NU_CUIT),11,'0'),3,8) || '-' ||
          SUBSTR(LPAD(TRIM(C.CJCM_NU_CUIT),11,'0'),11,1) AS "CUIT|string",
        C.CJCM_DI_UBICACION1 AS "UBICACION1|string",
        'COD POSTAL:' || C.CJCM_ZN_POSTAL || ' - ' || C.CJCM_DI_UBICACION2 AS "UBICACION2|string"
      FROM CART_ORDENES_PROCESOS OP
      JOIN CART_USUARIOS U
        ON U.CAUS_CD_USUARIO = OP.CAOS_CD_USUARIO
      JOIN CAJT_COMPANIAS C
        ON C.CJCM_CD_COMPANIA = U.CAUS_CAGR_CD_COMPANIA
      WHERE OP.CAOS_NU_SECUENCIA = :NU_SECUENCIA
    `,

    RAMOS: `
      WITH P AS (
        SELECT
          TRUNC(LEAST(CAOS_FE_A, CAOS_FE_B))    AS FD,
          TRUNC(GREATEST(CAOS_FE_A, CAOS_FE_B)) AS FH,
          NVL(CAOS_CARP_CD_RAMO,0)              AS RAMO_N
        FROM CART_ORDENES_PROCESOS
        WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA

        RP.CARP_DE_RAMO AS "DE_RAMO|string"
      FROM BATB_REASEG_RRC R
      JOIN CART_RAMOS_POLIZAS RP
        ON RP.CARP_CD_RAMO = R.RERR_CACE_CD_RAMO

      WHERE R.RERR_CJOD_FE_EFECTIVA >= P.FD
        AND R.RERR_CJOD_FE_EFECTIVA <  (P.FH + 1)
        AND (
              (P.RAMO_N = 0 AND R.RERR_CACE_CD_RAMO IN (1,8,18,19,20,21,22,23,24,25,26,50))

    `,

    Q_TOT_INVALIDADAS: `

          NVL(CAOS_CARP_CD_RAMO,0)              AS RAMO_N

        AND OD.CAOD_FE_OPERACION >= P.FD

    Q_INVALIDADAS_DETALLE: `
      WITH P AS (
        SELECT
          TRUNC(LEAST(CAOS_FE_A, CAOS_FE_B))    AS FD,
          TRUNC(GREATEST(CAOS_FE_A, CAOS_FE_B)) AS FH,
          NVL(CAOS_CARP_CD_RAMO,0)              AS RAMO_N,
          999999                                AS MAX_INV
        FROM CART_ORDENES_PROCESOS
        WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
      ),
      BASE AS (
        SELECT
          OD.CAOD_FE_OPERACION                  AS FECHA,
          OD.CAOD_CAPO_NU_POLIZA                AS POLIZA,
          OD.CAOD_CACE_NU_CERTIFICADO           AS CERTIF,
          CASE WHEN NVL(CE.CACE_NU_ENDOSO,0)=1 THEN 0 ELSE NVL(CE.CACE_NU_ENDOSO,0) END AS ENDOSO,
          CE.CACE_FE_DESDE                      AS DESDE,
          CE.CACE_FE_PROXIMA_FACTURACION        AS HASTA
        FROM CART_OPERACIONES_DIARIAS OD
        JOIN CART_CERTIFICADOS CE
          ON CE.CACE_CASU_CD_SUCURSAL = OD.CAOD_CASU_CD_SUCURSAL
         AND CE.CACE_CARP_CD_RAMO     = OD.CAOD_CARP_CD_RAMO
         AND CE.CACE_CAPO_NU_POLIZA   = OD.CAOD_CAPO_NU_POLIZA
         AND CE.CACE_NU_CERTIFICADO   = OD.CAOD_CACE_NU_CERTIFICADO
        CROSS JOIN P
        WHERE OD.CAOD_CAOP_CD_OPERACION = 1014
          AND OD.CAOD_FE_OPERACION >= P.FD
          AND OD.CAOD_FE_OPERACION <  (P.FH + 1)
          AND (
                (P.RAMO_N = 0 AND OD.CAOD_CARP_CD_RAMO IN (1,8,18,19,20,21,22,23,24,25,26,50))
             OR (P.RAMO_N <> 0 AND OD.CAOD_CARP_CD_RAMO = P.RAMO_N)

        CERTIF                      AS "CERTIF|int|center",
        ENDOSO                      AS "ENDOSO|int|center",
        TO_CHAR(DESDE,'DD/MM/YYYY') AS "DESDE|string|center",
        TO_CHAR(HASTA,'DD/MM/YYYY') AS "HASTA|string|center"
      FROM (
        SELECT B.*
        FROM BASE B
        WHERE ROWNUM <= (SELECT MAX_INV FROM P)
      )
      ORDER BY FECHA, POLIZA, CERTIF
    `,

    Q_EMISION_MONEDA: `

          NVL(CAOS_CARP_CD_RAMO,0)              AS RAMO_N

        CASE MONEDA
          WHEN '01' THEN 'PESOS'
          WHEN '02' THEN 'DOLARES'
          ELSE MONEDA
        END AS "MONEDA|string"
      FROM F

    NVL(R.RERR_CARG_SUP,0)+

       OR (P.RAMO_N <> 0 AND R.RERR_CACE_CD_RAMO = P.RAMO_N)

SELECT
  TO_CHAR(FECHA,'DD/MM/YYYY')                  AS "FECHA|string|center",
  POLIZA                                       AS "POLIZA|int|center",
  CERTIF                                       AS "CERTIF|int|center",
  CASE WHEN ENDOSO = 1 THEN 0 ELSE ENDOSO END AS "ENDOSO|int|center",
  ASEGURADO                                    AS "ASEGURADO|string|start",
  SUMA                                         AS "SUMA|money|end",
  TO_CHAR(DESDE,'DD/MM/YYYY')                  AS "DESDE|string|center",
  TO_CHAR(HASTA,'DD/MM/YYYY')                  AS "HASTA|string|center",
  COMISION                                     AS "COMISION|money|end",
  PRIMA                                        AS "PRIMA|money|end",
  DER_EM                                       AS "DER_EM|money|end",
  RECARGOS                                     AS "RECARGOS|money|end",
  IMPUESTOS                                    AS "IMPUESTOS|money|end",
  SELLADOS                                     AS "SELLADOS|money|end",
  BONIF                                        AS "BONIF|money|end",
  IIBB                                         AS "IIBB|money|end",
  PREMIO                                       AS "PREMIO|money|end"
FROM (
  SELECT B.*
  FROM BASE B
  WHERE ROWNUM <= (SELECT MAX_DET FROM P)
)
ORDER BY
  RAMO,
  SUCURSAL,
  MONEDA,
  POLIZA,
  CERTIF,
  FECHA,
  DESDE
