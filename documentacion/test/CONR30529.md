           CART_GERENCIAS

        AND CAGR_CD_COMPANIA = CAUS_CAGR_CD_COMPANIA
        AND CAGR_CD_UNIDAD_ADM = CAUS_CAGR_CD_UNIDAD_ADM
    `,

Q_RERR: `
WITH P AS (
  SELECT 
    TRUNC(GREATEST(CAOS_FE_A, CAOS_FE_B)) AS FECHA_HASTA
  FROM CART_ORDENES_PROCESOS
  WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
),

BASE AS (
  SELECT /*+ INDEX(R RERR_INDEX_1) */
    C.CARP_DE_RAMO                 AS DE_RAMO,
    R.RERR_CACE_CD_RAMO            AS RAMO,
    R.RERR_CACE_CAPO_NU_POLIZA     AS POLIZA,
    R.RERR_CACE_NU_CERTIFICADO     AS CERTIF,
    R.RERR_CJOD_FE_EFECTIVA        AS FE_EFECTIVA,
    R.RERR_CACE_FE_DESDE           AS FE_DESDE,
    R.RERR_CACE_FE_HASTA           AS FE_HASTA,
    R.RERR_CARE_MT_PRIMA           AS PRIMA,
    R.RERR_CARG_ADM + R.RERR_CARG_DEROP AS ADM_DEROP,
    R.RERR_MT_PRIMA_RECARGO        AS PMA_REC,
    R.RERR_CARE_MT_COMISION        AS COMISION,
    R.RERR_MT_PRIMA_CEDIDA         AS PMA_CED,
    R.RERR_NU_DIAS_VIGENCIA        AS DIAS_VIG,
    R.RERR_NU_DIAS_NO_CORRIDOS     AS DIAS_NO_COR,
    R.RERR_MT_PRIMA_REC_NO_DEVENG  AS PMA_REC_NDV,
    R.RERR_MT_RRC_NETA             AS RRC_NETA
  FROM BATB_REASEG_RRC R
  JOIN CART_RAMOS_POLIZAS C
    ON C.CARP_CD_RAMO = R.RERR_CACE_CD_RAMO
  CROSS JOIN P
  WHERE R.RERR_CJOD_FE_EFECTIVA 
        BETWEEN ADD_MONTHS(P.FECHA_HASTA, -1) AND P.FECHA_HASTA
    AND R.RERR_NU_DIAS_NO_CORRIDOS > 0
    AND R.RERR_CARE_MT_REC_ADM NOT IN (19,50)
)

-- DETALLE
SELECT
  DE_RAMO                      AS "DE_RAMO|string|start",
  RAMO                         AS "RAMO|int|center",
  POLIZA                       AS "POLIZA|int|center",
  CERTIF                       AS "CERTIF|int|center",
  FE_EFECTIVA                  AS "FE_EFECTIVA|date|center",
  FE_DESDE                     AS "FE_DESDE|date|center",
  FE_HASTA                     AS "FE_HASTA|date|center",
  PRIMA                        AS "PRIMA|money|end",
  ADM_DEROP                    AS "ADM_DEROP|money|end",
  PMA_REC                      AS "PMA_REC|money|end",
  COMISION                     AS "COMISION|money|end",
  PMA_CED                      AS "PMA_CED|money|end",
  DIAS_VIG                     AS "DIAS_VIG|int|center",
  DIAS_NO_COR                  AS "DIAS_NO_COR|int|center",
  PMA_REC_NDV                  AS "PMA_REC_NDV|money|end",
  RRC_NETA                     AS "RRC_NETA|money|end"
FROM BASE

UNION ALL

-- TOTAL POR RAMO
SELECT
  'TOTAL RAMO ' || RAMO,
  RAMO,
  NULL, NULL, NULL, NULL, NULL,
  SUM(PRIMA),
  SUM(ADM_DEROP),
  SUM(PMA_REC),
  SUM(COMISION),
  SUM(PMA_CED),
  SUM(DIAS_VIG),
  SUM(DIAS_NO_COR),
  SUM(PMA_REC_NDV),
  SUM(RRC_NETA)
FROM BASE
GROUP BY RAMO

UNION ALL

-- TOTAL GENERAL
SELECT
  'TOTAL GENERAL',
  NULL,
  NULL, NULL, NULL, NULL, NULL,
  SUM(PRIMA),
  SUM(ADM_DEROP),
  SUM(PMA_REC),
  SUM(COMISION),
  SUM(PMA_CED),
  SUM(DIAS_VIG),
  SUM(DIAS_NO_COR),
  SUM(PMA_REC_NDV),
  SUM(RRC_NETA)
FROM BASE

ORDER BY 2, 3 NULLS LAST
`
