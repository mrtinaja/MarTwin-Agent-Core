          SELECT      
                CJFA_DE_COMPROBANTE AS "Comprobante|string|center",
                nvl(SUM(CJFC_MT_CONCEPTO_TOTAL),0) AS "Mt Neto|money|center", 
                nvl(SUM(CJFC_MT_CONCEPTO_IVA),0) AS "Mt IVA|money|center", 
                nvl(SUM(CJFC_MT_IMPUESTO),0) AS "Mt Otros Imp|money|center",
                nvl(SUM(CJFC_MT_CONCEPTO_TOTAL),0) +
                nvl(SUM(CJFC_MT_CONCEPTO_IVA),0) +
                nvl(SUM(CJFC_MT_IMPUESTO),0)  AS "Total|money|center"
          FROM CAJT_COMPROBANTES, CAJT_COMPROBANTES_CONCEPTOS
          WHERE  CJFA_CJEG_NU_EGRESO IS NOT NULL
            AND CJFA_CAPS_NU_CUIT = CJFC_CAPS_NU_CUIT
            AND CJFA_NU_SUCURSAL = CJFC_CJFA_NU_SUCURSAL
            AND CJFA_NU_COMPROBANTE = CJFC_CJFA_NU_COMPROBANTE
          AND TO_CHAR(CJFA_CJEG_NU_EGRESO) = TO_CHAR(:NU_EGRESO)
            GROUP BY CJFA_NU_SUCURSAL,  
                CJFA_NU_COMPROBANTE, 
                CJFA_DE_COMPROBANTE, 
                CJFA_FE_COMPROBANTE, 
                CJFA_CATP_TP_PROVEEDOR,
                CJFA_CJEG_NU_EGRESO,
                CJFA_CD_COMPROBANTE,
                CJFA_NU_SUCURSAL,
                CJFA_NU_COMPROBANTE,
                CJFA_CAPS_NU_CUIT
`,

  Q_DATOS: `
SELECT
  /* ===== COMPAÑÍA ===== */
  CJCM.CJCM_CD_COMPANIA                     AS "Compania|int|center",
  CJCM.CJCM_CD_RIF                          AS "Rif|string|center",
  denl10000.get_nm_seguro(CJEG.CJEG_FE_ELABORACION,'N')
                                             AS "CompaniaNombre|string|start",

  CJCM.CJCM_NU_CUIT                         AS "CompaniaCuit|string|center",
  CJCM.CJCM_NU_INSCRIPCION_IB               AS "CompaniaIngBrutos|string|start",

  /* ===== BENEFICIARIO ===== */

  CAPS.CAPS_TP_CUENTA                       AS "TipoCuenta|string|start",
  CTCU.DESC_TIPO_CUENTA                     AS "TipoCuentaDesc|string|start",
  CAPS.CAPS_NU_CUENTA                       AS "NroCuenta|string|center",

  SUBSTR(CJCC.CJCC_DE_CONCEPTO_GANANCIA, 1, 25)
                                             AS "ConceptoGanancia|string|start"

FROM CAJT_EGRESOS CJEG
JOIN CAJT_COMPANIAS CJCM
  ON CJCM.CJCM_CD_COMPANIA = CJEG.CJEG_CJCM_CD_COMPANIA

LEFT JOIN CAJT_CONCEPTOS_GANANCIAS CJCC
  ON CJCC.CJCC_CD_CONCEPTO_GANANCIA = NVL(CJEG.CJEG_CJCC_CD_CONCEPTO_GANANCIA, 0)

LEFT JOIN CART_MONEDAS CAMO
  ON CAMO.CAMO_CD_MONEDA = CJEG.CJEG_CAMO_CD_MONEDA

LEFT JOIN CART_PROVEEDORES_SERVICIOS CAPS
  ON CAPS.CAPS_NU_CUIT = CJEG.CJEG_NU_CUIT
 AND CAPS.CAPS_CATP_TP_PROVEEDOR = CJEG.CJEG_CATP_TP_PROVEEDOR

LEFT JOIN CART_CIUDADES CACI
  ON CACI.CACI_CD_CIUDAD = CAPS.CAPS_CACI_CD_CIUDAD

LEFT JOIN (
  SELECT
    CTCU_TP_CUENTA,
    MAX(DECODE(CTCU_TP_CUENTA,'Q','CBU',CTCU_NM_CUENTA) || ': ') AS DESC_TIPO_CUENTA
  FROM CART_TIPOS_CUENTAS
  WHERE CTCU_IN_CUENTA IN ('C','D')
    AND CTCU_ST_CUENTA = 1
  GROUP BY CTCU_TP_CUENTA
) CTCU
  ON CTCU.CTCU_TP_CUENTA = TRIM(CAPS.CAPS_TP_CUENTA)

WHERE
  CJEG.CJEG_CJCM_CD_COMPANIA = :CD_COMPANIA
  AND CJEG.CJEG_CASU_CD_SUCURSAL = :CD_SUCURSAL
  AND CJEG.CJEG_NU_EGRESO = :NU_EGRESO
    `,
