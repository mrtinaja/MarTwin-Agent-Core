SELECT
  SRSS.SRSS_NU_ORDEN

  SRSS.SRSS_MDEF_CD_REGLA
    AS "Cod. Regla|string|start",

  SRSS.SRSS_NU_SCORE_REGLA
    AS "Score|int|end",

  LISTAGG(
      SVRS.SVRS_MDEF_CD_REGLA || ': ' ||
      NVL(TO_CHAR(SVRS.SVRS_NU_VALOR_REGLA),
          SVRS.SVRS_NU_VALOR_REGLA),
      '; '
  ) WITHIN GROUP (ORDER BY SVRS.SVRS_MDEF_CD_REGLA)
    AS "Lista datos|string|start"

FROM SINT_REGLAS_SCORE_X_SINIESTRO SRSS
LEFT JOIN SINT_VALOR_REGLA_X_SINIESTRO SVRS
  ON SVRS.SVRS_CASU_CD_SUCURSAL = SRSS.SRSS_CASU_CD_SUCURSAL
 AND SVRS.SVRS_SISI_NU_SINIESTRO = SRSS.SRSS_SISI_NU_SINIESTRO
 AND SVRS.SVRS_MDEF_CD_REGLA = SRSS.SRSS_MDEF_CD_REGLA

WHERE SRSS.SRSS_CASU_CD_SUCURSAL = :CD_SUCURSAL
  AND SRSS.SRSS_SISI_NU_SINIESTRO = :NU_SINIESTRO
  AND SRSS.SRSS_NU_SCORE_REGLA <> 0

GROUP BY
  SRSS.SRSS_NU_ORDEN,
  SRSS.SRSS_MDEF_CD_REGLA,
  SRSS.SRSS_NU_SCORE_REGLA

ORDER BY SRSS.SRSS_NU_ORDEN
