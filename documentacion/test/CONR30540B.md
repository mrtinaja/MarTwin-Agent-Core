export default {
  name: 'CONR30541B',
  effectiveDate: '2026-04-20',
  description: 'Listado de Totales de Emisión - Matriz paginada dinámica estilo RDF',

  params: {
    NU_SECUENCIA: { type: 'number', required: true }
  },

  queries: {
    MAIN: `
      SELECT 1 AS "ORDEN|int"
      FROM DUAL
    `,

    FECHAS: `
      SELECT
        TO_CHAR(CAOS_FE_A, 'DD/MM/YYYY') AS "FECHA_DESDE_TXT|string",
        TO_CHAR(CAOS_FE_B, 'DD/MM/YYYY') AS "FECHA_HASTA_TXT|string"
      FROM CART_ORDENES_PROCESOS
      WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
    `,

    PAGES: `
WITH PARAMS AS (
  SELECT
    TRUNC(LEAST(CAOS_FE_A, CAOS_FE_B)) AS F_DESDE,
    TRUNC(GREATEST(CAOS_FE_A, CAOS_FE_B)) AS F_HASTA,
    TO_CHAR(TRUNC(LEAST(CAOS_FE_A, CAOS_FE_B)), 'YYYYMM') AS P_AXI
  FROM CART_ORDENES_PROCESOS
  WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
),

UNIVERSO AS (
  SELECT
    R.RERR_CACE_CD_RAMO           AS RAMO,
    R.RERR_CARE_MT_PRIMA          AS PRIMA,
    R.RERR_CARG_ADM               AS ADM,
    R.RERR_CARG_DEROP             AS DEROP,
    R.RERR_CARG_FIN               AS FIN,
    R.RERR_CARG_INT               AS INT_INT,
    R.RERR_CARG_ISS               AS ISS,
    R.RERR_CARG_IVA               AS IVA,
    R.RERR_CARG_SLL               AS SLL,
    R.RERR_CARG_SUP               AS SUP,
    R.RERR_CARG_BON               AS BON,
    R.RERR_CARG_INGBR             AS IIBB,
    R.RERR_CARG_OTR               AS OTROS,
    R.RERR_CARE_MT_COMISION       AS COMISION,
    R.RERR_CARE_MT_PREMIO         AS PREMIO,
    R.RERR_CJOD_FE_EFECTIVA       AS FE,
    R.RERR_CJOD_CD_OPERACION      AS OP,
    R.RERR_CACE_CAPO_NU_POLIZA    AS POL,
    R.RERR_CACE_NU_CERTIFICADO    AS CERT,
    R.RERR_CARE_NU_ENDOSO         AS ENDOSO
  FROM BATB_REASEG_RRC R
  JOIN PARAMS P
    ON R.RERR_CJOD_FE_EFECTIVA >= P.F_DESDE
   AND R.RERR_CJOD_FE_EFECTIVA < P.F_HASTA + 1
  WHERE R.RERR_CJOD_CD_OPERACION NOT IN (2040, 2041, 7001, 7002, 7003, 7004)
    AND R.RERR_CACE_CD_SUCURSAL = 1
),

AXI AS (
  SELECT
    A.AXIF_RERR_CACE_CD_RAMO          AS RAMO,
    A.AXIF_RERR_CJOD_FE_EFECTIVA      AS FE,
    A.AXIF_RERR_CJOD_CD_OPERACION     AS OP,
    A.AXIF_RERR_CACE_CAPO_NU_POLIZA   AS POL,
    A.AXIF_RERR_CACE_NU_CERTIFICADO   AS CERT,
    A.AXIF_RERR_CARE_NU_ENDOSO        AS ENDOSO,
    SUM(A.AXIF_PRIMA_AJUSTADA_CIERRE)    AS PRIMA_AJ,
    SUM(A.AXIF_COMISION_AJUSTADA_CIERRE) AS COMISION_AJ
  FROM CONT_AXI_FACTURACION A
  JOIN PARAMS P
    ON A.AXIF_PERIODO = P.P_AXI
   AND A.AXIF_RERR_CJOD_FE_EFECTIVA >= P.F_DESDE
   AND A.AXIF_RERR_CJOD_FE_EFECTIVA < P.F_HASTA + 1
  GROUP BY
    A.AXIF_RERR_CACE_CD_RAMO,
    A.AXIF_RERR_CJOD_FE_EFECTIVA,
    A.AXIF_RERR_CJOD_CD_OPERACION,
    A.AXIF_RERR_CACE_CAPO_NU_POLIZA,
    A.AXIF_RERR_CACE_NU_CERTIFICADO,
    A.AXIF_RERR_CARE_NU_ENDOSO
),

BASE AS (
  SELECT
    U.RAMO                                 AS RAMO,
    SUM(U.PRIMA)                           AS PRIMA,
    SUM(U.ADM)                             AS ADM,
    SUM(U.DEROP)                           AS DEROP,
    SUM(U.FIN)                             AS FIN,
    SUM(U.INT_INT)                         AS INT_INT,
    SUM(U.ISS)                             AS ISS,
    SUM(U.IVA)                             AS IVA,
    SUM(U.SLL)                             AS SLL,
    SUM(U.SUP)                             AS SUP,
    SUM(U.BON)                             AS BON,
    SUM(U.IIBB)                            AS IIBB,
    SUM(U.OTROS)                           AS OTROS,
    SUM(U.COMISION)                        AS COMISION,
    SUM(U.PREMIO)                          AS PREMIO,
    SUM(NVL(A.PRIMA_AJ, 0))                AS PRIMA_AJ,
    SUM(NVL(A.COMISION_AJ, 0))             AS COMISION_AJ
  FROM UNIVERSO U
  LEFT JOIN AXI A
    ON U.RAMO   = A.RAMO
   AND U.FE     = A.FE
   AND U.OP     = A.OP
   AND U.POL    = A.POL
   AND U.CERT   = A.CERT
   AND U.ENDOSO = A.ENDOSO
  GROUP BY U.RAMO
  HAVING
       SUM(U.PRIMA) <> 0
    OR SUM(U.COMISION) <> 0
    OR SUM(NVL(A.PRIMA_AJ, 0)) <> 0
    OR SUM(NVL(A.COMISION_AJ, 0)) <> 0
),

RAMOS AS (
  SELECT
    B.RAMO,
    ROW_NUMBER() OVER (ORDER BY B.RAMO) AS RN
  FROM BASE B
),

PAGINADO AS (
  SELECT
    R.RAMO,
    CEIL(R.RN / 7) AS PAGINA,
    ROW_NUMBER() OVER (
      PARTITION BY CEIL(R.RN / 7)
      ORDER BY R.RAMO
    ) AS POS
  FROM RAMOS R
),

MAX_PAG AS (
  SELECT MAX(PAGINA) AS MAX_PAGINA
  FROM PAGINADO
),

TOTALES AS (
  SELECT
    SUM(B.PRIMA)        AS PRIMA,
    SUM(B.ADM)          AS ADM,
    SUM(B.DEROP)        AS DEROP,
    SUM(B.FIN)          AS FIN,
    SUM(B.INT_INT)      AS INT_INT,
    SUM(B.ISS)          AS ISS,
    SUM(B.IVA)          AS IVA,
    SUM(B.SLL)          AS SLL,
    SUM(B.SUP)          AS SUP,
    SUM(B.BON)          AS BON,
    SUM(B.IIBB)         AS IIBB,
    SUM(B.OTROS)        AS OTROS,
    SUM(B.COMISION)     AS COMISION,
    SUM(B.PREMIO)       AS PREMIO,
    SUM(B.PRIMA_AJ)     AS PRIMA_AJ,
    SUM(B.COMISION_AJ)  AS COMISION_AJ
  FROM BASE B
),

CONCEPTOS AS (
  SELECT  1 AS ORDEN, 'Prima'        AS ETIQUETA, 'PRIMA'       AS CAMPO FROM DUAL UNION ALL
  SELECT  2,           'Rec.Adm.',                  'ADM'                  FROM DUAL UNION ALL
  SELECT  3,           'Der.Emi.',                  'DEROP'                FROM DUAL UNION ALL
  SELECT  4,           'Rec.Fin.',                  'FIN'                  FROM DUAL UNION ALL
  SELECT  5,           'Imp.Int.',                  'INT_INT'              FROM DUAL UNION ALL
  SELECT  6,           'Iss',                       'ISS'                  FROM DUAL UNION ALL
  SELECT  7,           'Iva',                       'IVA'                  FROM DUAL UNION ALL
  SELECT  8,           'Sellados',                  'SLL'                  FROM DUAL UNION ALL
  SELECT  9,           'Super',                     'SUP'                  FROM DUAL UNION ALL
  SELECT 10,           'Bonific.',                  'BON'                  FROM DUAL UNION ALL
  SELECT 11,           'IIBB.',                     'IIBB'                 FROM DUAL UNION ALL
  SELECT 12,           'Otros',                     'OTROS'                FROM DUAL UNION ALL
  SELECT 13,           'Comision',                  'COMISION'             FROM DUAL UNION ALL
  SELECT 14,           'Premio',                    'PREMIO'               FROM DUAL UNION ALL
  SELECT 15,           'Prima Aj.',                 'PRIMA_AJ'             FROM DUAL UNION ALL
  SELECT 16,           'Comision Aj.',              'COMISION_AJ'          FROM DUAL
),

HEADER_HTML AS (
  SELECT
    P.PAGINA,
    '<tr><th>Ramo</th>' ||
    LISTAGG('<th>' || TO_CHAR(P.RAMO) || '</th>', '')
      WITHIN GROUP (ORDER BY P.POS) ||
    CASE
      WHEN P.PAGINA = M.MAX_PAGINA THEN '<th>Totales</th>'
      ELSE ''
    END ||
    '</tr>' AS HTML
  FROM PAGINADO P
  CROSS JOIN MAX_PAG M
  GROUP BY P.PAGINA, M.MAX_PAGINA
),

FILAS_HTML AS (
  SELECT
    C.ORDEN,
    P.PAGINA,
    '<tr><td>' || C.ETIQUETA || '</td>' ||

    DBMS_XMLGEN.CONVERT(
      RTRIM(
        XMLAGG(
          XMLELEMENT(
            E,
            '<td>' ||
            NVL(
              TO_CHAR(
                CASE C.CAMPO
                  WHEN 'PRIMA'       THEN B.PRIMA
                  WHEN 'ADM'         THEN B.ADM
                  WHEN 'DEROP'       THEN B.DEROP
                  WHEN 'FIN'         THEN B.FIN
                  WHEN 'INT_INT'     THEN B.INT_INT
                  WHEN 'ISS'         THEN B.ISS
                  WHEN 'IVA'         THEN B.IVA
                  WHEN 'SLL'         THEN B.SLL
                  WHEN 'SUP'         THEN B.SUP
                  WHEN 'BON'         THEN B.BON
                  WHEN 'IIBB'        THEN B.IIBB
                  WHEN 'OTROS'       THEN B.OTROS
                  WHEN 'COMISION'    THEN B.COMISION
                  WHEN 'PREMIO'      THEN B.PREMIO
                  WHEN 'PRIMA_AJ'    THEN B.PRIMA_AJ
                  WHEN 'COMISION_AJ' THEN B.COMISION_AJ
                END,
                'FM999G999G999G990D00',
                'NLS_NUMERIC_CHARACTERS='',.'''
              ),
              ' '
            ) ||
            '</td>'
          )
          ORDER BY P.POS
        ).EXTRACT('//text()').getClobVal(),
        CHR(10)
      ),
      1
    ) ||

    CASE
      WHEN P.PAGINA = M.MAX_PAGINA THEN
        '<td>' ||
        NVL(
          TO_CHAR(
            CASE C.CAMPO
              WHEN 'PRIMA'       THEN (SELECT T.PRIMA FROM TOTALES T)
              WHEN 'ADM'         THEN (SELECT T.ADM FROM TOTALES T)
              WHEN 'DEROP'       THEN (SELECT T.DEROP FROM TOTALES T)
              WHEN 'FIN'         THEN (SELECT T.FIN FROM TOTALES T)
              WHEN 'INT_INT'     THEN (SELECT T.INT_INT FROM TOTALES T)
              WHEN 'ISS'         THEN (SELECT T.ISS FROM TOTALES T)
              WHEN 'IVA'         THEN (SELECT T.IVA FROM TOTALES T)
              WHEN 'SLL'         THEN (SELECT T.SLL FROM TOTALES T)
              WHEN 'SUP'         THEN (SELECT T.SUP FROM TOTALES T)
              WHEN 'BON'         THEN (SELECT T.BON FROM TOTALES T)
              WHEN 'IIBB'        THEN (SELECT T.IIBB FROM TOTALES T)
              WHEN 'OTROS'       THEN (SELECT T.OTROS FROM TOTALES T)
              WHEN 'COMISION'    THEN (SELECT T.COMISION FROM TOTALES T)
              WHEN 'PREMIO'      THEN (SELECT T.PREMIO FROM TOTALES T)
              WHEN 'PRIMA_AJ'    THEN (SELECT T.PRIMA_AJ FROM TOTALES T)
              WHEN 'COMISION_AJ' THEN (SELECT T.COMISION_AJ FROM TOTALES T)
            END,
            'FM999G999G999G990D00',
            'NLS_NUMERIC_CHARACTERS='',.'''
          ),
          ' '
        ) ||
        '</td>'
      ELSE ''
    END ||

    '</tr>' AS HTML
  FROM CONCEPTOS C
  JOIN PAGINADO P
    ON 1 = 1
  JOIN BASE B
    ON B.RAMO = P.RAMO
  CROSS JOIN MAX_PAG M
  GROUP BY
    C.ORDEN,
    C.ETIQUETA,
    C.CAMPO,
    P.PAGINA,
    M.MAX_PAGINA
),

TABLAS_POR_PAGINA AS (
  SELECT
    H.PAGINA,
    '<div class="matrix-page">' ||
      '<table>' ||
        '<thead>' || H.HTML || '</thead>' ||
        '<tbody>' ||
          (
            SELECT DBMS_XMLGEN.CONVERT(
              RTRIM(
                XMLAGG(
                  XMLELEMENT(E, F.HTML)
                  ORDER BY F.ORDEN
                ).EXTRACT('//text()').getClobVal(),
                CHR(10)
              ),
              1
            )
            FROM FILAS_HTML F
            WHERE F.PAGINA = H.PAGINA
          ) ||
        '</tbody>' ||
      '</table>' ||
    '</div>' AS HTML
  FROM HEADER_HTML H
)

SELECT
  RTRIM(
    XMLAGG(
      XMLELEMENT(
        E,
        P.HTML ||
        CASE
          WHEN P.PAGINA < (SELECT MAX(X.PAGINA) FROM TABLAS_POR_PAGINA X)
          THEN '<<<page>>>'
          ELSE ''
        END
      )
      ORDER BY P.PAGINA
    ).getClobVal(),
    CHR(10)
  ) AS "HTML|string"
FROM TABLAS_POR_PAGINA P
`
  },
