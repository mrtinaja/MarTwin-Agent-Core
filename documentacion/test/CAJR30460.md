    Q_DETAILS: `
WITH P AS (
  SELECT
    TRUNC(MIN(LEAST(CAOS_FE_A, CAOS_FE_B)))     AS FECHA_DESDE,
    TRUNC(MAX(GREATEST(CAOS_FE_A, CAOS_FE_B))) AS FECHA_HASTA
  FROM CART_ORDENES_PROCESOS
  WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
)
SELECT
  TO_CHAR(CJEG.CJEG_FE_ELABORACION,'DD-MM-YYYY') AS "Fecha|string|center",
  CJEG.CJEG_NU_EGRESO                            AS "Egreso|int|center",
  SISI.SISI_NU_SINIESTRO                         AS "Siniestro|int|center",
  CJEG.CJEG_MT_TOTAL                             AS "Importe|money|end",

  SUBSTR(RVS.RV_MEANING,1,25)                    AS "Status Siniestro|string|center",
  SUBSTR(RVE.RV_MEANING,1,25)                    AS "Status Egreso|string|center",
  DECODE(
  CJEG.CJEG_CJTE_CD_EGRESO,
  700, 'CHEQUE',
  705, 'TRANSFERENCIA',
  715, 'NET BANKING',
  TO_CHAR(CJEG.CJEG_CJTE_CD_EGRESO)
) AS "Tipo Pago|string|center"

FROM P
JOIN CAJT_EGRESOS CJEG
  ON CJEG.CJEG_CJCM_CD_COMPANIA = 1
 AND CJEG.CJEG_CASU_CD_SUCURSAL = 1
 AND CJEG.CJEG_FE_ELABORACION >= NVL(P.FECHA_DESDE, DATE '1900-01-01')
 AND CJEG.CJEG_FE_ELABORACION <  NVL(P.FECHA_HASTA + 1, DATE '2999-12-31')

JOIN SINT_MOVIMIENTOS_EGRESOS SIME
  ON SIME.SIME_CJEG_CJCM_CD_COMPANIA = CJEG.CJEG_CJCM_CD_COMPANIA
 AND SIME.SIME_CJEG_CASU_CD_SUCURSAL = CJEG.CJEG_CASU_CD_SUCURSAL
 AND SIME.SIME_CJEG_NU_EGRESO        = CJEG.CJEG_NU_EGRESO

JOIN SINT_SINIESTROS SISI
  ON SISI.SISI_CASU_CD_SUCURSAL = SIME.SIME_SISI_CASU_CD_SUCURSAL
 AND SISI.SISI_NU_SINIESTRO     = SIME.SIME_SISI_NU_SINIESTRO

LEFT JOIN CG_REF_CODES RVS
  ON RVS.RV_DOMAIN = 'SINT_SINIESTROS.SISI_ST_SINIESTRO'
 AND TO_NUMBER(RVS.RV_LOW_VALUE) = SISI.SISI_ST_SINIESTRO

LEFT JOIN CG_REF_CODES RVE
  ON RVE.RV_DOMAIN = 'CAJT_EGRESOS.CJEG_ST_EGRESO'
 AND TO_NUMBER(RVE.RV_LOW_VALUE) = CJEG.CJEG_ST_EGRESO

LEFT JOIN CG_REF_CODES RVP
  ON RVP.RV_DOMAIN = 'CAJT_EGRESOS.CJEG_CJTE_CD_EGRESO'
 AND TO_NUMBER(RVP.RV_LOW_VALUE) = CJEG.CJEG_CJTE_CD_EGRESO

WHERE
  SISI.SISI_ST_SINIESTRO <> 21
  AND NOT EXISTS (
    SELECT 1
    FROM CAJT_CHEQUES CH
    WHERE CH.CJCH_CJCM_CD_COMPANIA = CJEG.CJEG_CJCM_CD_COMPANIA
      AND CH.CJCH_CASU_CD_SUCURSAL = CJEG.CJEG_CASU_CD_SUCURSAL
      AND CH.CJCH_CJEG_NU_EGRESO   = CJEG.CJEG_NU_EGRESO
  )
      
ORDER BY
  CJEG.CJEG_FE_ELABORACION DESC,
  CJEG.CJEG_NU_EGRESO DESC
FETCH FIRST 1 ROWS ONLY

    `,
    CABECERA: `
  SELECT
    TO_CHAR(SYSDATE,'DD/MM/YYYY') AS "FECHA_REPORTE|string"
  FROM DUAL
`
