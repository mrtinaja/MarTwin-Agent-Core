export default {
  name: 'SINR35000',
  effectiveDate: '2026-01-01',
  description: 'Nota formal de requerimiento de documentación',

  params: {
    CASU_CD_SUCURSAL: { type: 'number', required: true },
    NRO_SINIESTRO: { type: 'number', required: true },
    FE_SOLICITUD: { type: 'date', required: true }
  },

  queries: {
    MAIN: `
      SELECT 75 AS "ORDEN" FROM DUAL
    `,

    Q_POLIZA_CERTIFICADO: `
      SELECT
        sice_sisi_casu_cd_sucursal  AS "CASU_SUCURSAL|int",
        sice_sisi_nu_siniestro      AS "NU_SINIESTRO|int",
        sice_cace_casu_cd_sucursal  AS "SUCURSAL|int",
        sice_cace_carp_cd_ramo      AS "RAMO|int",
        sice_cace_capo_nu_poliza    AS "POLIZA|int",
        sice_cace_nu_certificado    AS "CERTIFICADO|int",
        sisi_fe_ocurrencia          AS "FE_SINIESTRO|date",
        :FE_SOLICITUD               AS "FE_SOL_REQ|date"
      FROM sint_certificados_siniestros,
           sint_siniestros
      WHERE sisi_casu_cd_sucursal = sice_sisi_casu_cd_sucursal
        AND sisi_nu_siniestro     = sice_sisi_nu_siniestro
        AND sice_sisi_casu_cd_sucursal = :CASU_CD_SUCURSAL
        AND sice_sisi_nu_siniestro     = :NRO_SINIESTRO
    `,

    Q_CERTIFICADO: `
SELECT
    rp.CARP_DE_RAMO                       AS "RAMO_DESC|string",
    c.CACE_CAPO_NU_POLIZA                 AS "POLIZA|int",
    c.CACE_NU_CERTIFICADO                 AS "CERTIFICADO|int",
    p.CAPU_IN_COLECTIVO                   AS "COLECTIVO|string",

    NVL(c.CACE_CACN_TDOC_TOMADOR,
        c.CACE_CACN_CD_NACIONALIDAD)      AS "TIPO_DOC|string",

    NVL(c.CACE_CACN_NDOC_TOMADOR,
        c.CACE_CACN_NU_CEDULA_RIF)        AS "NRO_DOC|string",

    cl.CACN_NM_APELLIDO_RAZON             AS "ASEGURADO|string",
    cl.CACN_FE_NACIMIENTO                 AS "FE_NACIMIENTO|date",
    cl.CACN_DI_COBRO1                     AS "DOMICILIO|string",
    ci.CACI_DE_CIUDAD                     AS "CIUDAD|string",
    es.CAES_DE_ESTADO                     AS "ESTADO|string",
    cl.CACN_ZN_POSTAL_COBRO               AS "CP|string",
    cl.CACN_NU_TELEFONO_COBRO             AS "TELEFONO|string",

    TRUNC(SYSDATE)                        AS "FECHA_CARTA|date"

FROM sint_certificados_siniestros sc

JOIN CART_CERTIFICADOS c
  ON c.CACE_CARP_CD_RAMO     = sc.sice_cace_carp_cd_ramo
 AND c.CACE_CAPO_NU_POLIZA   = sc.sice_cace_capo_nu_poliza
 AND c.CACE_NU_CERTIFICADO   = sc.sice_cace_nu_certificado
 AND c.CACE_CASU_CD_SUCURSAL = sc.sice_cace_casu_cd_sucursal

JOIN CART_CLIENTES cl
  ON cl.CACN_CD_NACIONALIDAD = c.CACE_CACN_CD_NACIONALIDAD
 AND cl.CACN_NU_CEDULA_RIF   = c.CACE_CACN_NU_CEDULA_RIF

JOIN CART_RAMOS_POLIZAS rp
  ON rp.CARP_CD_RAMO = c.CACE_CARP_CD_RAMO

JOIN CART_PRODUCTOS p
  ON p.CAPU_CARP_CD_RAMO = c.CACE_CARP_CD_RAMO
 AND p.CAPU_CD_PRODUCTO  = c.CACE_CAPU_CD_PRODUCTO

LEFT JOIN CART_ESTADOS es
  ON es.CAES_CD_ESTADO = cl.CACN_CAES_CD_ESTADO_COB

LEFT JOIN CART_CIUDADES ci
  ON ci.CACI_CAES_CD_ESTADO = cl.CACN_CAES_CD_ESTADO_COB
 AND ci.CACI_CD_CIUDAD      = cl.CACN_CACI_CD_CIUDAD_COB

WHERE sc.sice_sisi_casu_cd_sucursal = :CASU_CD_SUCURSAL
  AND sc.sice_sisi_nu_siniestro     = :NRO_SINIESTRO
`,

    Q_REQUERIMIENTOS: `
SELECT
    r.SIRQ_DE_REQUERIMIENTO       AS "DESCRIPCION|string"

FROM SINT_REQUERIMIENTOS_SINIESTROS s

JOIN SINT_REQUERIMIENTOS r
  ON r.SIRQ_CD_REQUERIMIENTO = s.SIRI_SIRQ_CD_REQUERIMIENTO

WHERE s.SIRI_SISI_CASU_CD_SUCURSAL = :CASU_CD_SUCURSAL
  AND s.SIRI_SISI_NU_SINIESTRO     = :NRO_SINIESTRO

ORDER BY s.SIRI_SIRQ_CD_REQUERIMIENTO
    `,

    Q_COMPANIA: `
      SELECT
        CJCM_CD_COMPANIA AS "CD_COMPANIA|string",
        CJCM_DI_UBICACION1 AS "LINEA1|string",
        CJCM_ZN_POSTAL || ' - ' || CJCM_DI_UBICACION2 AS "LINEA2|string",
        'TEL.: ' || RTRIM(LTRIM(CJCM_NU_TELEFONO1)) AS "LINEA3|string",
        'SERVICIO DE ATENCION A CLIENTES: ' ||
        RTRIM(LTRIM(CJCM_NU_TELEFONO2)) AS "LINEA4|string"
      FROM CAJT_COMPANIAS
    `
  },
