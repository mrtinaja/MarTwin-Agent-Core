      SELECT
        S.SISI_NU_SINIESTRO AS SINIESTRO,
        CE.SICE_CACE_CAPO_NU_POLIZA AS POLIZA,
        CE.SICE_CACE_NU_CERTIFICADO AS CERTIFICADO,

        TO_CHAR(S.SISI_FE_DECLARACION,'DD-MM-YYYY') AS FECHA_DECLARACION,
        TO_CHAR(S.SISI_FE_OCURRENCIA,'DD-MM-YYYY')  AS FECHA_SINIESTRO,

        NVL(S.SISI_DI_MAIL,' ')       AS MAIL,
        NVL(S.SISI_NU_TELEF_ADIC,' ') AS TELEFONO,

        NVL(E.CAES_DE_ESTADO,' ')     AS PROVINCIA,
        NVL(C.CACI_DE_CIUDAD,' ')     AS CIUDAD,

        NVL(S.SISI_DE_LUGAR_OCURRENCIA,' ') AS DOMICILIO,

        NVL(TO_CHAR(S.SISI_CD_POSTAL_OCURRENCIA),' ') AS COD_POSTAL_DEN,
        NVL(S.SISI_DI_MAIL_ADIC,' ')                   AS MAIL_ADIC,

        NVL(TO_CHAR(S.SISI_MT_DANIO_TOTAL),' ') AS MONTO,

        NVL(S.SISI_DE_SINIESTRO,' ') AS HECHOS,
        NVL(S.SISI_DE_DAÑOS,' ')     AS DANOS,

        NVL(CS.SICS_DE_CAUSA,' ') AS CAUSA,
        'NO' AS FAST_TRACK,

        NVL(U.CAUS_NM_USUARIO,' ')   AS ANALISTA,
        NVL(R.CARP_DE_RAMO,' ')      AS RAMO

      FROM SINT_SINIESTROS S

      JOIN SINT_CERTIFICADOS_SINIESTROS CE
        ON CE.SICE_SISI_CASU_CD_SUCURSAL = S.SISI_CASU_CD_SUCURSAL
       AND CE.SICE_SISI_NU_SINIESTRO     = S.SISI_NU_SINIESTRO

      LEFT JOIN CART_ESTADOS E
        ON E.CAES_CD_ESTADO = S.SISI_CACI_CAES_CD_ESTADO

      LEFT JOIN CART_CIUDADES C
        ON C.CACI_CD_CIUDAD = S.SISI_CACI_CD_CIUDAD

      LEFT JOIN SINT_CAUSA_SINIESTROS CS
        ON CS.SICS_CARP_CD_RAMO = S.SISI_CARP_CD_RAMO
       AND CS.SICS_CD_CAUSA_SINIESTRO = S.SISI_SICS_CD_CAUSA_SINIESTRO

      LEFT JOIN CART_RAMOS_POLIZAS R
        ON R.CARP_CD_RAMO = CE.SICE_CACE_CARP_CD_RAMO

      JOIN CART_USUARIOS U
        ON U.CAUS_CD_USUARIO = S.SISI_CAAN_CD_ANALISTA

      WHERE S.SISI_CASU_CD_SUCURSAL = :SUCURSAL
        AND S.SISI_NU_SINIESTRO     = :SINIESTRO
    `,

    Q_2: `
      SELECT
        NVL(CB.CACB_DE_COBERTURA,' ')         AS COBERTURA,
        NVL(TO_CHAR(RC.SICC_MT_RESERVA),' ') AS RESERVA,
        NVL(IDE.SIIA_DE_ITEM_AFECTADO,' ')   AS ITEM

      FROM SINT_RESERVA_COBERTURA_CERTIFI RC

      JOIN CART_COBERTURAS CB
        ON CB.CACB_CD_COBERTURA = RC.SICC_SICO_CACB_CD_COBERTURA
       AND CB.CACB_CARB_CD_RAMO = RC.SICC_SICO_CACB_CARB_CD_RAMO

      JOIN SINT_ITEMS_AFECTADOS_SINIESTRO I
        ON I.SIAS_CD_SUCURSAL  = RC.SICC_SISI_CASU_CD_SUCURSA
       AND I.SIAS_NU_SINIESTRO = RC.SICC_SICO_SISI_NU_SINIESTRO
       AND I.SIAS_CD_RAMO      = RC.SICC_SICO_CACB_CARB_CD_RAMO
       AND I.SIAS_CD_COBERTURA = RC.SICC_SICO_CACB_CD_COBERTURA

      JOIN SINT_ITEMS_AFECTADOS IDE
        ON IDE.SIIA_CD_ITEM_AFECTADO = I.SIAS_CD_ITEM_AFECTADO

      WHERE RC.SICC_SISI_CASU_CD_SUCURSA   = :SUCURSAL
        AND RC.SICC_SICO_SISI_NU_SINIESTRO = :SINIESTRO
    `,

    Q_3: `
      SELECT
        NVL(C.CACN_NM_APELLIDO_RAZON,' ') AS ASEGURADO,
        NVL(C.CACN_DI_COBRO1,' ')        AS DOMICILIO,
        NVL(CI.CACI_DE_CIUDAD,' ')       AS CIUDAD,
        NVL(E.CAES_DE_ESTADO,' ')        AS PROVINCIA,
        NVL(TO_CHAR(C.CACN_ZN_POSTAL_COBRO),' ') AS COD_POSTAL,
        NVL(C.CACN_NU_TELEFONO_COBRO,' ') AS TELEFONO

      FROM SINT_SINIESTROS S
      JOIN CART_CLIENTES C
        ON C.CACN_CD_NACIONALIDAD = S.SISI_CACN_CD_NACIONALIDAD
       AND C.CACN_NU_CEDULA_RIF   = S.SISI_CACN_NU_CEDULA_RIF

      LEFT JOIN CART_CIUDADES CI
        ON CI.CACI_CD_CIUDAD = C.CACN_CACI_CD_CIUDAD_COB
       AND CI.CACI_CAES_CD_ESTADO = C.CACN_CAES_CD_ESTADO_COB

      LEFT JOIN CART_ESTADOS E
        ON E.CAES_CD_ESTADO = C.CACN_CAES_CD_ESTADO_COB

      WHERE S.SISI_CASU_CD_SUCURSAL = :SUCURSAL
        AND S.SISI_NU_SINIESTRO     = :SINIESTRO
    `,
