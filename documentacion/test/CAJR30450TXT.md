export default {
  name: "CAJR30450TXT",
  effectiveDate: "2025-01-01",
  description: "Listado de Egresos (por NU_SECUENCIA)",

  params: {
    NU_SECUENCIA: { type: "number", required: true },
  },

  queries: {
    MAIN: `
      SELECT 75 AS "ORDEN"
      FROM CART_ORDENES_PROCESOS
      WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
    `,

Q_EGRESOS: `
WITH P AS (
  SELECT
    TRUNC(LEAST(CAOS_FE_A, CAOS_FE_B))    AS F_DESDE,
    TRUNC(GREATEST(CAOS_FE_A, CAOS_FE_B)) AS F_HASTA,
    CAOS_CACN_CD_NACIONALIDAD             AS TIPO_PARAM,
    CASE
      WHEN CAOS_CACN_CD_NACIONALIDAD IN ('S','A') THEN 'S'
      WHEN CAOS_CACN_CD_NACIONALIDAD = 'P' THEN 'N'
      ELSE 'N'
    END AS EGR_SINIESTROS
  FROM CART_ORDENES_PROCESOS
  WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
),

/* SINIESTROFORMULA */
S AS (
  SELECT
    SIME_CJEG_NU_EGRESO,
    MIN(SIME_SISI_NU_SINIESTRO) AS SIME_SISI_NU_SINIESTRO
  FROM SINT_MOVIMIENTOS_EGRESOS
  WHERE SIME_CJEG_CJCM_CD_COMPANIA = 1
    AND SIME_CJEG_CASU_CD_SUCURSAL = 1
  GROUP BY SIME_CJEG_NU_EGRESO
),

/* CHEQUE */
C AS (
  SELECT
    CJCH_CJEG_NU_EGRESO,
    MIN(CJCH_NU_CHEQUE) AS CJCH_NU_CHEQUE
  FROM CAJT_CHEQUES
  GROUP BY CJCH_CJEG_NU_EGRESO
),

/* RECIBOS para POLIZA cuando tipo = REINTEGRO */
REC AS (
  SELECT
    CARE_CJEG_NU_EGRESO,
    MIN(CARE_CAPO_NU_POLIZA) AS CARE_CAPO_NU_POLIZA
  FROM CART_RECIBOS
  GROUP BY CARE_CJEG_NU_EGRESO
),

/* CERTIFICADOS SINIESTROS */
SC AS (
  SELECT
    SICE_SISI_NU_SINIESTRO,
    MIN(SICE_CACE_CAPO_NU_POLIZA)   AS SICE_CACE_CAPO_NU_POLIZA,
    MIN(SICE_CACE_CASU_CD_SUCURSAL) AS SICE_CACE_CASU_CD_SUCURSAL,
    MIN(SICE_CACE_CARP_CD_RAMO)     AS SICE_CACE_CARP_CD_RAMO,
    MIN(SICE_CACE_NU_CERTIFICADO)   AS SICE_CACE_NU_CERTIFICADO
  FROM SINT_CERTIFICADOS_SINIESTROS
  WHERE SICE_SISI_CASU_CD_SUCURSAL = 1
  GROUP BY SICE_SISI_NU_SINIESTRO
),

BASE AS (
  SELECT
    E.CJEG_CJTE_CD_EGRESO,
    E.CJEG_CJTE_CARP_CD_RAMO,
    E.CJEG_NM_BENEFICIARIO,
    E.CJEG_MT_TOTAL,
    E.CJEG_NU_EGRESO,
    E.CJEG_FE_ELABORACION,
    E.CJEG_FE_STATUS,
    E.CJEG_CADM_NU_CUENTA,
    E.CJEG_ST_EGRESO,
    E.CJEG_NU_CUIT,
    E.CJEG_NU_CEDULA_RIF,

    P.TIPO_PARAM,
    P.EGR_SINIESTROS,

    NVL(S.SIME_SISI_NU_SINIESTRO, 0) AS SINIESTRO_CALC,
    NVL(C.CJCH_NU_CHEQUE, 0)         AS CHEQUE_CALC,
    REC.CARE_CAPO_NU_POLIZA,
    SC.SICE_CACE_CAPO_NU_POLIZA,

    CASE
      WHEN P.EGR_SINIESTROS = 'N' THEN 'PROVEEDOR'
      WHEN E.CJEG_NU_CUIT IS NULL AND NVL(S.SIME_SISI_NU_SINIESTRO, 0) = 0 THEN 'REINTEGRO'
      WHEN E.CJEG_NU_CUIT IS NULL AND NVL(S.SIME_SISI_NU_SINIESTRO, 0) <> 0 THEN 'SINIESTRO'
      WHEN E.CJEG_NU_CUIT IS NOT NULL AND NVL(S.SIME_SISI_NU_SINIESTRO, 0) <> 0 AND P.EGR_SINIESTROS = 'S' THEN 'SINIESTRO'
      WHEN E.CJEG_NU_CUIT IS NOT NULL THEN 'PROVEEDOR'
      ELSE 'PROVEEDOR'
    END AS TIPO_CF1,

    CASE
      WHEN E.CJEG_NU_CUIT IS NOT NULL THEN TO_CHAR(E.CJEG_NU_CUIT)
      WHEN E.CJEG_NU_CUIT IS NULL THEN TO_CHAR(E.CJEG_NU_CEDULA_RIF)
      ELSE TO_CHAR(E.CJEG_NU_CUIT)
    END AS DNI_CUIT_CF2

  FROM CAJT_EGRESOS E
  CROSS JOIN P
  LEFT JOIN S
    ON S.SIME_CJEG_NU_EGRESO = E.CJEG_NU_EGRESO
  LEFT JOIN C
    ON C.CJCH_CJEG_NU_EGRESO = E.CJEG_NU_EGRESO
  LEFT JOIN REC
    ON REC.CARE_CJEG_NU_EGRESO = E.CJEG_NU_EGRESO
  LEFT JOIN SC
    ON SC.SICE_SISI_NU_SINIESTRO = NVL(S.SIME_SISI_NU_SINIESTRO, 0)
  WHERE
    E.CJEG_FE_ELABORACION >= P.F_DESDE
    AND E.CJEG_FE_ELABORACION < P.F_HASTA + 1
)

SELECT
  B.TIPO_CF1 AS "TIPO|string|start",
  B.CJEG_CJTE_CARP_CD_RAMO AS "Ramo|int|center",
  B.SINIESTRO_CALC AS "Siniestro|int|center",
  B.CJEG_NM_BENEFICIARIO AS "Beneficiario|string|start",
  B.CJEG_MT_TOTAL AS "Importe|money|end",
  B.CHEQUE_CALC AS "Cheque|int|center",
  B.CJEG_NU_EGRESO AS "Egreso|int|center",
  B.CJEG_FE_ELABORACION AS "Fecha|date|center",
  B.CJEG_FE_STATUS AS "Fecha Status|date|center",
  SUBSTR(R.RV_MEANING, 1, 10) AS "Estado Egr.|string|start",

  CASE
    WHEN B.CJEG_CJTE_CARP_CD_RAMO = 18 THEN
      NVL((
        SELECT MIN(V.VICE_VIDD_CD_SUCURSAL)
        FROM VIDT_CERTIFICADOS V
        JOIN SINT_CERTIFICADOS_SINIESTROS SCS
          ON SCS.SICE_SISI_CASU_CD_SUCURSAL = 1
         AND SCS.SICE_SISI_NU_SINIESTRO     = B.SINIESTRO_CALC
         AND V.VICE_CACE_CASU_CD_SUCURSAL   = SCS.SICE_CACE_CASU_CD_SUCURSAL
         AND V.VICE_CACE_CARP_CD_RAMO       = SCS.SICE_CACE_CARP_CD_RAMO
         AND V.VICE_CACE_CAPO_NU_POLIZA     = SCS.SICE_CACE_CAPO_NU_POLIZA
         AND V.VICE_CACE_NU_CERTIFICADO     = SCS.SICE_CACE_NU_CERTIFICADO
      ), 0)
    ELSE
      NVL((
        SELECT MIN(CC.CACE_CAZB_CD_SUCURSAL)
        FROM CART_CERTIFICADOS CC
        JOIN SINT_CERTIFICADOS_SINIESTROS SCS
          ON SCS.SICE_SISI_CASU_CD_SUCURSAL = 1
         AND SCS.SICE_SISI_NU_SINIESTRO     = B.SINIESTRO_CALC
         AND CC.CACE_CASU_CD_SUCURSAL       = SCS.SICE_CACE_CASU_CD_SUCURSAL
         AND CC.CACE_CARP_CD_RAMO           = SCS.SICE_CACE_CARP_CD_RAMO
         AND CC.CACE_CAPO_NU_POLIZA         = SCS.SICE_CACE_CAPO_NU_POLIZA
         AND CC.CACE_NU_CERTIFICADO         = SCS.SICE_CACE_NU_CERTIFICADO
      ), 0)
  END AS "Sucursal|int|center",

  B.CJEG_CADM_NU_CUENTA AS "CBU|string|center",

  CASE
    WHEN B.TIPO_CF1 = 'REINTEGRO' THEN NVL(B.CARE_CAPO_NU_POLIZA, 0)
    ELSE NVL(B.SICE_CACE_CAPO_NU_POLIZA, 0)
  END AS "Poliza|int|center",

  NVL((
    SELECT MAX(TO_CHAR(F.CJFA_NU_SUCURSAL) || TO_CHAR(F.CJFA_NU_COMPROBANTE))
    FROM CAJT_COMPROBANTES F
    WHERE TO_CHAR(F.CJFA_CAPS_NU_CUIT) = B.DNI_CUIT_CF2
       OR F.CJFA_CJEG_NU_EGRESO = B.CJEG_NU_EGRESO
  ), 'CCCC') AS "Factura|string|center",

  B.DNI_CUIT_CF2 AS "DNI/CUIT|string|center"

FROM BASE B
JOIN CG_REF_CODES R
  ON R.RV_LOW_VALUE = B.CJEG_ST_EGRESO
 AND R.RV_DOMAIN    = 'CAJT_EGRESOS.CJEG_ST_EGRESO'

WHERE
     (B.TIPO_PARAM = 'P' AND B.TIPO_CF1 = 'PROVEEDOR')
  OR (B.TIPO_PARAM = 'S' AND B.TIPO_CF1 = 'SINIESTRO')
  OR (B.TIPO_PARAM = 'A' AND B.TIPO_CF1 IN ('PROVEEDOR', 'SINIESTRO'))

ORDER BY B.CJEG_NU_EGRESO
`,
  },
