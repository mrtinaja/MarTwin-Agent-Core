export default {
  name: "CONR30527",
  effectiveDate: "2025-01-01",
  description: "Distribución geográfica de la producción",

  params: {
    NU_SECUENCIA: { type: "number", required: true },
  },

  queries: {
    
    MAIN: `SELECT 75 AS "ORDEN" FROM DUAL`,

    /* =========================
       CABECERA
       ========================= */
    CABECERA: `
      SELECT
        OP.CAOS_CD_USUARIO            AS "USUARIO|string",
        U.CAUS_CAGR_CD_UNIDAD_ADM     AS "CD_UNIDAD_ADM|string",
        G.CAGR_NM_UNIDAD_ADM          AS "NM_UNIDAD_ADM|string",
        TO_CHAR(SYSDATE,'DD/MM/YYYY') AS "FECHA_REP|string",
        TO_CHAR(SYSDATE,'HH24:MI:SS') AS "HORA_REP|string",
        'CONR30527'                   AS "REF|string",
        TO_CHAR(TRUNC(LEAST(OP.CAOS_FE_A, OP.CAOS_FE_B)),'DD/MM/YYYY')    AS "FECHA_DESDE|string",
        TO_CHAR(TRUNC(GREATEST(OP.CAOS_FE_A, OP.CAOS_FE_B)),'DD/MM/YYYY') AS "FECHA_HASTA|string"
      FROM CART_ORDENES_PROCESOS OP
      JOIN CART_USUARIOS U
        ON U.CAUS_CD_USUARIO = OP.CAOS_CD_USUARIO
      JOIN CART_GERENCIAS G
        ON G.CAGR_CD_COMPANIA   = U.CAUS_CAGR_CD_COMPANIA
       AND G.CAGR_CD_UNIDAD_ADM = U.CAUS_CAGR_CD_UNIDAD_ADM
      WHERE OP.CAOS_NU_SECUENCIA = :NU_SECUENCIA
    `,

    /* =========================
       Q_DISTRIBUCION
       - ROLLUP (totales reales)
       - “TOTAL provincia” sin repetir jurisdicción
       - Gris: se logra con <span class="tprov|tgen">...</span> en cada celda
       ========================= */
    Q_DISTRIBUCION: `
WITH P AS (
  SELECT
    TRUNC(LEAST(CAOS_FE_A, CAOS_FE_B))     AS FD,
    TRUNC(GREATEST(CAOS_FE_A, CAOS_FE_B)) AS FH
  FROM CART_ORDENES_PROCESOS
  WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
),

BASE AS (
  SELECT
    NVL(E.CAES_CD_ESTADO,999)          AS ORDEN_PROV,
    NVL(E.CAES_DE_ESTADO,'SIN ESTADO') AS JURISDICCION,
    NVL(RP.CARP_DE_RAMO,'SIN RAMO')    AS RAMO,

    SUM(CASE
      WHEN B.RERR_CJOD_CD_OPERACION IN (6001,6003,6004,2043,2044)
      THEN NVL(B.RERR_CARE_MT_PRIMA,0)
      ELSE 0
    END) AS PRIMAS,

    SUM(CASE
      WHEN B.RERR_CJOD_CD_OPERACION IN (6001,6003,6004,2043,2044)
      THEN NVL(B.RERR_CARG_ADM,0)
         + NVL(B.RERR_CARG_DEROP,0)
         + NVL(B.RERR_CARG_FIN,0)
      ELSE 0
    END) AS RECARGOS,

    SUM(CASE
      WHEN B.RERR_CJOD_CD_OPERACION IN (2002,2042,6002)
      THEN NVL(B.RERR_CARE_MT_PRIMA,0)
         + NVL(B.RERR_CARG_ADM,0)
         + NVL(B.RERR_CARG_DEROP,0)
         + NVL(B.RERR_CARG_FIN,0)
      ELSE 0
    END) AS ANULACIONES

  FROM BATB_REASEG_RRC B
  CROSS JOIN P

  JOIN CART_CERTIFICADOS C
    ON C.CACE_CASU_CD_SUCURSAL = B.RERR_CACE_CD_SUCURSAL
   AND C.CACE_CARP_CD_RAMO     = B.RERR_CACE_CD_RAMO
   AND C.CACE_CAPO_NU_POLIZA   = B.RERR_CACE_CAPO_NU_POLIZA
   AND C.CACE_NU_CERTIFICADO   = B.RERR_CACE_NU_CERTIFICADO

  JOIN CART_RAMOS_POLIZAS RP
    ON RP.CARP_CD_RAMO = B.RERR_CACE_CD_RAMO

  LEFT JOIN CART_ESTADOS E
    ON E.CAES_CD_ESTADO =
       DECODE(NVL(C.CACE_CD_ESTADO_SELLADO,1),0,1,NVL(C.CACE_CD_ESTADO_SELLADO,1))

  WHERE B.RERR_CJOD_FE_EFECTIVA >= P.FD
    AND B.RERR_CJOD_FE_EFECTIVA <  (P.FH + 1)
    AND B.RERR_CJOD_CD_OPERACION IN (6001,6003,6004,2043,2044,2002,2042,6002)

  GROUP BY
    NVL(E.CAES_CD_ESTADO,999),
    NVL(E.CAES_DE_ESTADO,'SIN ESTADO'),
    NVL(RP.CARP_DE_RAMO,'SIN RAMO')
),

AGRUPADO AS (
  SELECT
    ORDEN_PROV,
    JURISDICCION,
    RAMO,
    SUM(PRIMAS)      AS PRIMAS,
    SUM(RECARGOS)    AS RECARGOS,
    SUM(ANULACIONES) AS ANULACIONES,
    SUM(PRIMAS + RECARGOS - ANULACIONES) AS TOTAL,
    GROUPING(RAMO)         AS GR_RAMO,
    GROUPING(JURISDICCION) AS GR_JUR
  FROM BASE
  GROUP BY ROLLUP( (ORDEN_PROV, JURISDICCION), RAMO )
),

SALIDA AS (
  SELECT
    ORDEN_PROV,
    JURISDICCION,
    RAMO,
    PRIMAS, RECARGOS, ANULACIONES, TOTAL,
    CASE
      WHEN GR_RAMO = 1 AND GR_JUR = 0 THEN 1
      WHEN GR_JUR = 1 THEN 2
      ELSE 0
    END AS ES_TOTAL,
    GR_JUR,
    GR_RAMO
  FROM AGRUPADO
)

SELECT
 CASE
  WHEN ES_TOTAL = 1 THEN '<span class="tprov tleft">TOTAL '||JURISDICCION||'</span>'
  WHEN ES_TOTAL = 2 THEN '<span class="tgen tleft">TOTAL GENERAL</span>'
  ELSE JURISDICCION
END AS "JURISDICCION|string|start",

CASE
  WHEN ES_TOTAL = 1 THEN '<span class="tprov" style="display:block; height:11px;"> </span>'
  WHEN ES_TOTAL = 2 THEN '<span class="tgen" style="display:block; height:11px;"> </span>'
  ELSE RAMO
END AS "RAMO|string|start",

  /* Importes (formato + gris por span) */
  CASE
    WHEN ES_TOTAL = 1 THEN '<span class="tprov num">'||TO_CHAR(PRIMAS,'FM999G999G999G999G990D00')||'</span>'
    WHEN ES_TOTAL = 2 THEN '<span class="tgen  num">'||TO_CHAR(PRIMAS,'FM999G999G999G999G990D00')||'</span>'
    ELSE '<span class="num">'||TO_CHAR(PRIMAS,'FM999G999G999G999G990D00')||'</span>'
  END AS "PRIMAS|string|end",

  CASE
    WHEN ES_TOTAL = 1 THEN '<span class="tprov num">'||TO_CHAR(RECARGOS,'FM999G999G999G999G990D00')||'</span>'
    WHEN ES_TOTAL = 2 THEN '<span class="tgen  num">'||TO_CHAR(RECARGOS,'FM999G999G999G999G990D00')||'</span>'
    ELSE '<span class="num">'||TO_CHAR(RECARGOS,'FM999G999G999G999G990D00')||'</span>'
  END AS "RECARGOS|string|end",

  CASE
    WHEN ES_TOTAL = 1 THEN '<span class="tprov num">'||TO_CHAR(ANULACIONES,'FM999G999G999G999G990D00')||'</span>'
    WHEN ES_TOTAL = 2 THEN '<span class="tgen  num">'||TO_CHAR(ANULACIONES,'FM999G999G999G999G990D00')||'</span>'
    ELSE '<span class="num">'||TO_CHAR(ANULACIONES,'FM999G999G999G999G990D00')||'</span>'
  END AS "ANULACIONES|string|end",

  CASE
    WHEN ES_TOTAL = 1 THEN '<span class="tprov num">'||TO_CHAR(TOTAL,'FM999G999G999G999G990D00')||'</span>'
    WHEN ES_TOTAL = 2 THEN '<span class="tgen  num">'||TO_CHAR(TOTAL,'FM999G999G999G999G990D00')||'</span>'
    ELSE '<span class="num">'||TO_CHAR(TOTAL,'FM999G999G999G999G990D00')||'</span>'
  END AS "TOTAL|string|end"

FROM SALIDA
ORDER BY
  ORDEN_PROV,
  GR_JUR,
  GR_RAMO,
  RAMO
`
