export default {
  name: "CARR30436",
  effectiveDate: "2025-01-01",
  description: "Reporte Reaseguro - Cierre Mensual",

  params: {
    NU_SECUENCIA: { type: "number", required: true }
  },

  queries: {

    /* =========================
       MAIN
    ========================= */
    MAIN: `
      SELECT 75 AS "ORDEN"
      FROM DUAL
    `,

    /* =========================
       PERIODO DESDE SECUENCIA
    ========================= */
    P_BASE: `
      SELECT
        TRUNC(GREATEST(CAOS_FE_A, CAOS_FE_B),'MM')  AS F_DESDE,
        LAST_DAY(GREATEST(CAOS_FE_A, CAOS_FE_B))   AS F_HASTA
      FROM CART_ORDENES_PROCESOS
      WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
    `,

    /* =========================
       TOTAL POR RAMO (ONLINE)
    ========================= */
Q_RAMO: `
WITH P AS (
  SELECT
    TRUNC(GREATEST(CAOS_FE_A, CAOS_FE_B),'MM')  AS F_DESDE,
    LAST_DAY(GREATEST(CAOS_FE_A, CAOS_FE_B))   AS F_HASTA
  FROM CART_ORDENES_PROCESOS
  WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
),

BASE AS (
  SELECT /*+ INDEX(r RERR_INDEX_1) */
    r.rerr_cace_cd_ramo,
    r.rerr_care_mt_prima,
    r.rerr_carg_adm,
    r.rerr_carg_derop,
    r.rerr_mt_prima_recargo,
    r.rerr_care_mt_comision,
    r.rerr_mt_prima_cedida,
    r.rerr_mt_prima_rec_no_deveng,
    r.rerr_mt_reaseg_no_deveng,
    r.rerr_mt_rrc_neta,
    r.rerr_nu_dias_no_corridos,
    r.rerr_nu_dias_vigencia,
    r.rerr_care_mt_rec_adm
  FROM BATB_REASEG_RRC r
  CROSS JOIN P
  WHERE
    r.rerr_cjod_fe_efectiva >= P.F_DESDE
    AND r.rerr_cjod_fe_efectiva <  P.F_HASTA + 1
),

AGR AS (
  SELECT

    rerr_cace_cd_ramo AS RAMO,

    SUM(rerr_care_mt_prima) PRIMA,
    SUM(NVL(rerr_carg_adm,0)+NVL(rerr_carg_derop,0)) ADM_DEROP,
    SUM(rerr_mt_prima_recargo) PRIMA_REC,
    SUM(rerr_care_mt_comision) COMISION,
    SUM(rerr_mt_prima_cedida) PRIMA_CED,
    SUM(rerr_mt_prima_rec_no_deveng) PRIMA_REC_NDV,

    ROUND(
      SUM(
        rerr_care_mt_comision *
        rerr_nu_dias_no_corridos /
        NULLIF(rerr_nu_dias_vigencia,0)
      ),2
    ) COMISION_NDV,

    SUM(rerr_mt_reaseg_no_deveng) PRIMA_CED_NDV,
    SUM(rerr_mt_rrc_neta) RRC_NETA

  FROM BASE

  WHERE
    NOT (
      rerr_care_mt_rec_adm NOT IN (19,50)
      AND rerr_nu_dias_no_corridos <> 0
    )

  GROUP BY
    rerr_cace_cd_ramo
)

/* =========================
   Detalle + Total General
   ========================= */

SELECT
  RAMO,
  PRIMA,
  ADM_DEROP,
  COMISION,
  PRIMA_REC,
  PRIMA_CED,
  PRIMA_REC_NDV,
  COMISION_NDV,
  PRIMA_CED_NDV,
  RRC_NETA,
  0 RRC_PRIMA_AJ,
  0 RRC_COM_AJ,
  0 PRIMA_AJ,
  0 COMISION_AJ,
  1 ORDEN
FROM AGR

UNION ALL

SELECT
  9999 AS RAMO,        -- Marca TOTAL
  SUM(PRIMA),
  SUM(ADM_DEROP),
  SUM(COMISION),
  SUM(PRIMA_REC),
  SUM(PRIMA_CED),
  SUM(PRIMA_REC_NDV),
  SUM(COMISION_NDV),
  SUM(PRIMA_CED_NDV),
  SUM(RRC_NETA),
  0,
  0,
  0,
  0,
  2 ORDEN
FROM AGR

ORDER BY ORDEN, RAMO
`,

    /* =========================
       RRC MENSUAL
    ========================= */
    Q_RRC: `
      WITH P AS (
        SELECT
          TRUNC(GREATEST(CAOS_FE_A, CAOS_FE_B),'MM')  AS F_DESDE,
          LAST_DAY(GREATEST(CAOS_FE_A, CAOS_FE_B))   AS F_HASTA
        FROM CART_ORDENES_PROCESOS
        WHERE CAOS_NU_SECUENCIA = :NU_SECUENCIA
      )

      SELECT

        TRUNC(r.rerr_cjod_fe_efectiva,'MM')
          AS "MES|date|center",

        r.rerr_cace_cd_ramo
          AS "RAMO|string|center",

        SUM(r.rerr_care_mt_prima)
          AS "PRIMA|money|end",

        SUM(NVL(r.rerr_carg_adm,0)+NVL(r.rerr_carg_derop,0))
          AS "ADM_DEROP|money|end",

        SUM(r.rerr_mt_prima_recargo)
          AS "PRIMA_REC|money|end",

        SUM(r.rerr_care_mt_comision)
          AS "COMISION|money|end",

        SUM(r.rerr_mt_prima_cedida)
          AS "PRIMA_CED|money|end",

        SUM(r.rerr_mt_rrc_neta)
          AS "RRC_NETA|money|end"

      FROM BATB_REASEG_RRC r
      CROSS JOIN P

      WHERE
        r.rerr_cjod_fe_efectiva
          BETWEEN P.F_DESDE AND P.F_HASTA

        AND r.rerr_nu_dias_no_corridos <> 0

        AND r.rerr_care_mt_rec_adm NOT IN (19,50)

      GROUP BY
        TRUNC(r.rerr_cjod_fe_efectiva,'MM'),
        r.rerr_cace_cd_ramo

      ORDER BY 1,2
    `
  },
