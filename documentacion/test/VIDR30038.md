      VIDD_NM_SOLICITANTE,
      VIDD_TP_DOCUMENTO,
      VIDD_NU_DOCUMENTO,
      VIDD_NU_TELEFONO,
      VIDD_CAPJ_CD_SUCURSAL,

      VIOE_DI_LUGAR_EXAMEN,

      VIOE_IN_ORINA,
      VIOE_IN_ECG,
      VIOE_IN_RAYOS_X,
      VIOE_IN_HIV,
      VIOE_IN_SANGRE,
      VIOE_IN_EMBARAZADA,
      VIDD_DE_OBSERVACIONES_MEDICO

FROM  VIDT_DATOS_DECLARACIONES VIDD
JOIN  VIDT_ORDENES_EXAMEN VIOE
ON    VIOE.VIOE_VIDD_CD_SUCURSAL = VIDD.VIDD_CAPJ_CD_SUCURSAL
AND   VIOE.VIOE_VIDD_NU_DECLARACION = VIDD.VIDD_NU_DECLARACION

JOIN  CART_SUCURSAL_BANCO SUC
ON    SUC.CAPJ_CD_SUCURSAL = VIDD.VIDD_CAPJ_CD_SUCURSAL

WHERE VIDD.VIDD_CAPJ_CD_SUCURSAL = :cod_sucursal
AND   VIDD.VIDD_NU_DECLARACION = :nro_declaracion
`
