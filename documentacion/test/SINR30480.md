  description: "Orden de Liquidación Global",

  params: {
    SUCURSAL:  { type: "number", required: true },
    SINIESTRO: { type: "number", required: true }
  },

  queries: {

    MAIN: `
      SELECT 75 AS "ORDEN" FROM DUAL
    `,

    /* =========================
       Q_COMPANIA
    ========================= */
    Q_COMPANIA: `
      SELECT
        CJCM_DI_UBICACION1 AS "LINEA_1|string",
        CJCM_ZN_POSTAL || ' - ' || CJCM_DI_UBICACION2 AS "LINEA_2|string",
        'TEL.: ' || RTRIM(LTRIM(CJCM_NU_TELEFONO1)) ||
        ' - FAX.: ' || RTRIM(LTRIM(CJCM_NU_FAX)) AS "LINEA_3|string",
        'SERVICIO DE ATENCION A CLIENTES: ' ||
        RTRIM(LTRIM(CJCM_NU_TELEFONO2)) AS "LINEA_4|string",
        'I.V.A. RESPONSABLE INSCRIPTO - A.G. PERCEPCION R.G. 3125'
        AS "LINEA_5|string",
        'ING. BRUTOS: ' ||
        RTRIM(SUBSTR(LTRIM(TO_CHAR(CJCM_NU_INSCRIPCION_IB)),1,3)||'-'||
              SUBSTR(LTRIM(TO_CHAR(CJCM_NU_INSCRIPCION_IB)),4,5)||'-'||
              SUBSTR(LTRIM(TO_CHAR(CJCM_NU_INSCRIPCION_IB)),9,1))
        || '   C.U.I.T.: ' ||
        RTRIM(SUBSTR(LTRIM(TO_CHAR(CJCM_NU_CUIT)),1,2)||'-'||
              SUBSTR(LTRIM(TO_CHAR(CJCM_NU_CUIT)),3,8)||'-'||
              SUBSTR(LTRIM(TO_CHAR(CJCM_NU_CUIT)),11,1))
        AS "LINEA_6|string"
      FROM CAJT_COMPANIAS
      WHERE CJCM_CD_COMPANIA = 1
    `,

    /* =========================
       Q_LIQUIDACION
    ========================= */
    Q_LIQUIDACION: `
      SELECT
        SILI_SISI_NU_SINIESTRO          AS "SINIESTRO|int|center",
        SILI_NU_LIQUIDACION             AS "LIQ|int|center",
       CASE
            WHEN SILI_CJTE_CD_EGRESO IN (710,715)
              THEN 'PAGO DE HONORARIOS SINIESTRO'
            WHEN SILI_CJTE_CD_EGRESO IN (700,705)
              THEN 'PAGO DE INDEMNIZACION'
            ELSE 'PAGO'
          END

          || '<br/>' ||

          SILI_NM_BENEFICIARIO

          || '<br/>' ||

          'Net Banking : ' || CTCU_NM_CUENTA

          || '<br/>' ||

          DECODE(
              CTCU_IN_CUENTA,
              'C',
                SUBSTR(SILI_CADM_NU_CUENTA,1,3) || '-' ||
                LPAD(TO_CHAR(TO_NUMBER(SUBSTR(SILI_CADM_NU_CUENTA,4,10))),6,0) || '/' ||
                SUBSTR(SILI_CADM_NU_CUENTA,14,1),
              SILI_CADM_NU_CUENTA
          )

          AS "CONCEPTO|string|start",
        SILI_TP_PAGO                    AS "TIPO_PAGO|string|center",
       CASE
          WHEN SILI_MT_IVA != 0 THEN
              TO_CHAR(SILI_MT_PAGO, '999G999G999D00')
              || ' (' ||
              ROUND(
                (SILI_MT_IVA / NULLIF(SILI_MT_PAGO,0)) * 100
              ,0)
              || '%)'
          ELSE
              TO_CHAR(SILI_MT_PAGO, '999G999G999D00')
        END
        AS "IMPORTE|string|end",
        SILI_MT_IVA                     AS "IVA|money|end",
        SILI_FE_PAGO                    AS "FECHA|date|center",
        SILI_ST_LIQUIDACION             AS "STATUS|string|center",
         SILI_SISI_CASU_CD_SUCURSAL      AS "SUCURSAL|int|center"
      FROM SINH_LIQUIDACIONES SILI
      JOIN CART_MONEDAS
        ON CAMO_CD_MONEDA = SILI_CAPO_CD_MONEDA
      LEFT JOIN CART_TIPOS_CUENTAS CT
      ON CT.CTCU_TP_CUENTA = SILI_CADM_TP_CUENTA
      WHERE SILI_SISI_CASU_CD_SUCURSAL = :SUCURSAL
      AND   SILI_SISI_NU_SINIESTRO     = :SINIESTRO
    `,

   Q_TOTALES: `
SELECT
  NVL(SUM(
      CASE
        WHEN SILI_ST_LIQUIDACION = 'PEN'
        THEN SILI_MT_PAGO + SILI_MT_IVA
        ELSE 0
      END
  ),0) AS TOTAL_GENERAL_NUM,

  NVL(SUM(
      CASE
        WHEN SILI_ST_LIQUIDACION = 'PEN'
        THEN SILI_MT_PAGO + SILI_MT_IVA
        ELSE 0
      END
  ),0) AS "TOTAL_INDEMNIZACION|money|end",

  0 AS "TOTAL_GASTOS|money|end",

  NVL(SUM(
      CASE
        WHEN SILI_ST_LIQUIDACION = 'PEN'
        THEN SILI_MT_PAGO + SILI_MT_IVA
        ELSE 0
      END
  ),0) AS "TOTAL_GENERAL|money|end",

  RECTOR.F_ALETRAS(
      NVL(SUM(
        CASE
          WHEN SILI_ST_LIQUIDACION = 'PEN'
          THEN SILI_MT_PAGO + SILI_MT_IVA
          ELSE 0
        END
      ),0)
  ) || '.-' AS "IMPORTE_LETRAS|string"

FROM SINH_LIQUIDACIONES
WHERE SILI_SISI_CASU_CD_SUCURSAL = :SUCURSAL
AND   SILI_SISI_NU_SINIESTRO     = :SINIESTRO
`,

Q_AUTORIZANTES: `
SELECT *
FROM (
    SELECT DISTINCT
           A.SAUT_CD_USUARIO      AS "LEGAJO|string",
           C.RV_MEANING           AS "JERARQUIA|string",
           TO_CHAR(A.SAUT_FE_EFECTIVA,'DD-MON-YY') AS "FECHA|string"
    FROM SINT_SEGUIMIENTO_SINIESTROS S
    JOIN SINT_AUTORIZANTES A
      ON S.SISG_CD_USUARIO = A.SAUT_CD_USUARIO
    JOIN CG_REF_CODES C
      ON C.RV_LOW_VALUE = A.SAUT_CD_JERARQUIA
     AND C.RV_DOMAIN = 'JERARQUIA_LIQUIDACION_SINIESTRO'
    WHERE S.SISG_SISI_CASU_CD_SUCURSAL = :SUCURSAL
      AND S.SISG_SISI_NU_SINIESTRO     = :SINIESTRO
      AND S.SISG_CAOP_CD_OPERACION     = 3009
)
WHERE ROWNUM <= 4
`,

Q_CONCEPTO_LIQ: `
SELECT
  CASE
    WHEN CL.CACN_NM_APELLIDO_RAZON IS NULL
      OR SCS.SICE_CD_CLASE_PRODUCTO IS NULL
    THEN
        'SINIESTRO: ' || :SINIESTRO ||
        '  POLIZA: ' || NVL(SCS.SICE_CACE_CAPO_NU_POLIZA,0) ||
        '  CERTIFICADO: ' || NVL(SCS.SICE_CACE_NU_CERTIFICADO,0)
    ELSE
        'SINIESTRO: ' || :SINIESTRO ||
        '  POLIZA: ' || SCS.SICE_CACE_CAPO_NU_POLIZA ||
        '  CERTIFICADO: ' || SCS.SICE_CACE_NU_CERTIFICADO ||
        '  NOMBRE: ' || CL.CACN_NM_APELLIDO_RAZON ||
        '  DNI: ' || SI.SISI_CACN_NU_CEDULA_RIF ||
        '  PRODUCTO: ' || PR.CAPU_DE_PRODUCTO ||
        '  SUCURSAL N°: ' || SCS.SICE_NU_SUC_AFECTADA ||
        '  PRODUCTO N°: ' || SCS.SICE_NU_CTA_AFECTADA
  END AS "CONCEPTO_LIQ|string"
FROM SINT_CERTIFICADOS_SINIESTROS SCS
LEFT JOIN SINT_SINIESTROS SI
       ON SI.SISI_CASU_CD_SUCURSAL = SCS.SICE_SISI_CASU_CD_SUCURSAL
      AND SI.SISI_NU_SINIESTRO     = SCS.SICE_SISI_NU_SINIESTRO
LEFT JOIN CART_CLIENTES CL
       ON CL.CACN_CD_NACIONALIDAD = SI.SISI_CACN_CD_NACIONALIDAD
      AND CL.CACN_NU_CEDULA_RIF   = SI.SISI_CACN_NU_CEDULA_RIF
LEFT JOIN CART_PRODUCTOS PR
       ON PR.CAPU_CARP_CD_RAMO = SI.SISI_SICS_CARP_CD_RAMO
      AND PR.CAPU_CACY_CD_CLASE = SCS.SICE_CD_CLASE_PRODUCTO
WHERE SCS.SICE_SISI_CASU_CD_SUCURSAL = :SUCURSAL
  AND SCS.SICE_SISI_NU_SINIESTRO     = :SINIESTRO
`,

Q_CLIENTE_SELECT: `
SELECT
  CASE
    WHEN ICC.INCC_CD_RENTA = 6 THEN 'CLIENTE SELECT'
    ELSE NULL
  END AS "CLIENTE_SELECT|string"
FROM SINT_SINIESTROS SI
JOIN CART_CLIENTES CC
  ON CC.CACN_CD_NACIONALIDAD = SI.SISI_CACN_CD_NACIONALIDAD
 AND CC.CACN_NU_CEDULA_RIF   = SI.SISI_CACN_NU_CEDULA_RIF
JOIN INFOSEG.INFT_CLIENTES_CLASIFICADOS ICC
  ON CC.CACN_NU_NUP = ICC.INCC_NU_NUP
WHERE SI.SISI_CASU_CD_SUCURSAL = :SUCURSAL
  AND SI.SISI_NU_SINIESTRO     = :SINIESTRO
`
