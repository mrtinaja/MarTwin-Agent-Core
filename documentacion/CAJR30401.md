queries:{


Q_DATOS:`
SELECT

denl10000.get_nm_seguro(CJEG.CJEG_FE_ELABORACION,'N')
"COMPANIA|string",

CJCM.CJCM_DI_UBICACION1
"DIR1|string",

CJCM.CJCM_DI_UBICACION2
"DIR2|string",

CJCM.CJCM_NU_TELEFONO1
"TEL|string",

CJCM.CJCM_NU_FAX
"FAX|string",

CJEG.CJEG_NU_EGRESO
"EGRESO|int",

CAPS.CAPS_NU_CUIT
"CUIT|string",

CAPS.CAPS_NM_PROVEEDOR
"PROVEEDOR|string",

CJEG.CJEG_NM_BENEFICIARIO
"BENEFICIARIO|string",

CAPS.CAPS_DI_OFICINA1
"DIR_BEN1|string",

CAPS.CAPS_DI_OFICINA2
"DIR_BEN2|string",

CACI.CACI_DE_CIUDAD
"CIUDAD|string",

CAPS.CAPS_CAZP_CD_POSTAL
"CP|string",

CJEG.CJEG_MT_TOTAL
"IMPORTE|money",

RECTOR.F_ALETRAS(NVL(CJEG.CJEG_MT_TOTAL,0))
"IMPORTE_LETRAS|string",

CTCU.CTCU_NM_CUENTA
AS "TIPO_CUENTA|string",

CAPS.CAPS_NU_CUENTA
AS "CUENTA|string"

FROM CAJT_EGRESOS CJEG

JOIN CAJT_COMPANIAS CJCM
ON CJCM.CJCM_CD_COMPANIA = CJEG.CJEG_CJCM_CD_COMPANIA

LEFT JOIN CART_PROVEEDORES_SERVICIOS CAPS
ON CAPS.CAPS_NU_CUIT = CJEG.CJEG_NU_CUIT
AND CAPS.CAPS_CATP_TP_PROVEEDOR = CJEG.CJEG_CATP_TP_PROVEEDOR

LEFT JOIN CART_CIUDADES CACI
ON CACI.CACI_CD_CIUDAD = CAPS.CAPS_CACI_CD_CIUDAD

LEFT JOIN (
  SELECT *
  FROM cajt_cheques
  WHERE cjch_cjcm_cd_compania = :CD_COMPANIA
  AND cjch_casu_cd_sucursal = :CD_SUCURSAL
  AND cjch_cjeg_nu_egreso = :NU_EGRESO
  AND ROWNUM = 1
) ch
  ON ch.cjch_cjcm_cd_compania = cjeg.cjeg_cjcm_cd_compania
 AND ch.cjch_casu_cd_sucursal = cjeg.cjeg_casu_cd_sucursal
 AND ch.cjch_cjeg_nu_egreso = cjeg.cjeg_nu_egreso


LEFT JOIN cajt_cuentas_bancarias cu
  ON cu.cjcb_cd_cuenta = ch.cjch_cjcb_cd_cuenta
 AND cu.cjcb_casu_cd_sucursal = ch.cjch_casu_cd_sucursal
 AND cu.cjcb_cjcm_cd_compania = ch.cjch_cjcm_cd_compania

LEFT JOIN CART_TIPOS_CUENTAS CTCU
  ON CTCU.CTCU_TP_CUENTA = CAPS.CAPS_TP_CUENTA

WHERE
CJEG.CJEG_CJCM_CD_COMPANIA = :CD_COMPANIA
AND CJEG.CJEG_CASU_CD_SUCURSAL = :CD_SUCURSAL
AND CJEG.CJEG_NU_EGRESO = :NU_EGRESO
`,



Q_COMPROBANTES:`
SELECT

CJFA_CD_COMPROBANTE
|| ' ' ||
LPAD(CJFA_NU_SUCURSAL,9,'0')
|| ' - ' ||
LPAD(CJFA_NU_COMPROBANTE,10,'0')

"COMPROBANTE|string",

NVL(SUM(CJFC_MT_CONCEPTO_TOTAL),0)
"NETO|money",

NVL(SUM(CJFC_MT_CONCEPTO_IVA),0)
"IVA|money",

NVL(SUM(CJFC_MT_IMPUESTO),0)
"OTROS|money",

NVL(SUM(CJFC_MT_CONCEPTO_TOTAL),0)
+
NVL(SUM(CJFC_MT_CONCEPTO_IVA),0)
+
NVL(SUM(CJFC_MT_IMPUESTO),0)

"MONTO|money"

FROM CAJT_COMPROBANTES CJFA

JOIN CAJT_COMPROBANTES_CONCEPTOS CJFC
ON CJFA_CAPS_NU_CUIT = CJFC_CAPS_NU_CUIT
AND CJFA_NU_SUCURSAL = CJFC_CJFA_NU_SUCURSAL
AND CJFA_NU_COMPROBANTE = CJFC_CJFA_NU_COMPROBANTE

WHERE CJFA_CJEG_NU_EGRESO = :NU_EGRESO

GROUP BY
CJFA_CD_COMPROBANTE,
CJFA_NU_SUCURSAL,
CJFA_NU_COMPROBANTE
`,



Q_TOTAL_COMPROBANTES:`
SELECT

NVL(SUM(CJFC_MT_CONCEPTO_TOTAL),0)
+
NVL(SUM(CJFC_MT_CONCEPTO_IVA),0)
+
NVL(SUM(CJFC_MT_IMPUESTO),0)

"TOTAL|money"

FROM CAJT_COMPROBANTES CJFA

JOIN CAJT_COMPROBANTES_CONCEPTOS CJFC
ON CJFA_CAPS_NU_CUIT = CJFC_CAPS_NU_CUIT
AND CJFA_NU_SUCURSAL = CJFC_CJFA_NU_SUCURSAL
AND CJFA_NU_COMPROBANTE = CJFC_CJFA_NU_COMPROBANTE

WHERE CJFA_CJEG_NU_EGRESO = :NU_EGRESO
`,



Q_RETENCIONES: `
SELECT

CJCR_DE_CONCEPTO_RETENCION
AS "DESCRIPCION|string",

TO_CHAR(CJER_PO_ALICUOTA,'FM9990.00')
AS "ALICUOTA|string",

CJER_MT_RETENCION
AS "MONTO|money"

FROM CAJT_EGRESOS_RETENCIONES CJER

JOIN CAJT_CONCEPTOS_RETENCIONES CJCR
ON CJER_CD_CONCEPTO_RETENCION = CJCR_CD_CONCEPTO_RETENCION

WHERE CJER_CJEG_NU_EGRESO = :NU_EGRESO
`,



Q_TOTAL_RETENCIONES:`
SELECT
NVL(SUM(CJER_MT_RETENCION),0)
"TOTAL|money"
FROM CAJT_EGRESOS_RETENCIONES
WHERE CJER_CJEG_NU_EGRESO = :NU_EGRESO
`
